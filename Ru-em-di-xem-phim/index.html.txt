  <script>
    const noTexts = [
      "Hông bé ơi 😜",
      "Kó cái giề mà Kó :3",
      "Ai cho mà bấm hả?",
      "Cấm từ chối nhaaa 🥺",
      "Nút này bị hỏng rùi!",
      "Mơ đii, hổng cho bấm đâu 😝",
      "Chạy đâu cho thoát nà",
      "Bấm CÓ lẹ lên điii ❤️",
      "Chỉ được chọn CÓ thui!",
      "Năn nỉ bấm Có mừ 🥰"
    ];

    let textIdx = 0;
    let yesScale = 1;
    let isCentered = false;

    const btnNo = document.getElementById('btnNo');
    const btnYes = document.getElementById('btnYes');
    const btnGroup = document.getElementById('btnGroup');
    const mainQuestion = document.getElementById('mainQuestion');
    const successBox = document.getElementById('successBox');
    const faceNormal = document.getElementById('faceNormal');
    const faceHappy = document.getElementById('faceHappy');
    const bunnyWrap = document.getElementById('bunnyWrap');

    // Âm thanh Web Audio API (không cần tải file ngoài)
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    
    function playBoing() {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(300, audioCtx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(700, audioCtx.currentTime + 0.15);
      gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.15);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.15);
    }

    function playSuccessMelody() {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const notes = [523.25, 659.25, 783.99, 1046.50];
      notes.forEach((freq, i) => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime + i * 0.12);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime + i * 0.12);
        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + i * 0.12 + 0.35);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start(audioCtx.currentTime + i * 0.12);
        osc.stop(audioCtx.currentTime + i * 0.12 + 0.35);
      });
    }

    // Hiệu ứng né tránh & Đưa Có vào giữa
    function dodgeNoButton() {
      playBoing();

      if (!isCentered) {
        btnYes.classList.add('center-state');
        isCentered = true;
      }

      yesScale += 0.12;
      btnYes.style.setProperty('--btn-scale', yesScale);

      // Đổi chữ
      btnNo.innerText = noTexts[textIdx % noTexts.length];
      textIdx++;

      // Nhảy sang vị trí ngẫu nhiên
      btnNo.style.position = 'fixed';
      const pad = 80;
      const maxX = window.innerWidth - btnNo.offsetWidth - pad;
      const maxY = window.innerHeight - btnNo.offsetHeight - pad;
      const randomX = Math.max(pad, Math.random() * maxX);
      const randomY = Math.max(pad, Math.random() * maxY);

      btnNo.style.left = `${randomX}px`;
      btnNo.style.top = `${randomY}px`;
    }

    btnNo.addEventListener('mouseenter', dodgeNoButton);
    btnNo.addEventListener('touchstart', (e) => {
      e.preventDefault();
      dodgeNoButton();
    });

    // Khi bấm CÓ
    btnYes.addEventListener('click', () => {
      playSuccessMelody();

      // Đổi mặt thỏ & nhảy múa
      faceNormal.style.display = 'none';
      faceHappy.style.display = 'block';
      bunnyWrap.classList.add('happy-dance');

      // Ẩn câu hỏi & nút
      mainQuestion.style.display = 'none';
      btnGroup.style.display = 'none';
      btnNo.style.display = 'none';

      // Hiện chữ chốt kèo
      successBox.style.display = 'flex';

      // Pháo hoa trái tim bùng nổ
      confetti({
        particleCount: 120,
        spread: 100,
        origin: { y: 0.6 },
        colors: ['#ff3366', '#ff758c', '#ffd4de', '#ffffff']
      });
    });

    // Tim bay nền
    setInterval(() => {
      const heart = document.createElement('div');
      heart.className = 'heart-bg';
      heart.innerHTML = '💖';
      heart.style.left = Math.random() * 100 + 'vw';
      heart.style.fontSize = Math.random() * 15 + 14 + 'px';
      heart.style.animationDuration = Math.random() * 3 + 5 + 's';
      document.body.appendChild(heart);
      setTimeout(() => heart.remove(), 7000);
    }, 600);
  </script>
